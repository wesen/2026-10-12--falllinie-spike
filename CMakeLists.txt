cmake_minimum_required(VERSION 3.16)
project(falllinie_spike LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
find_package(Threads REQUIRED)
add_library(terrain src/terrain.cpp)
target_include_directories(terrain PUBLIC src)
target_compile_options(terrain PRIVATE -Wall -Wextra)
add_executable(test-terrain tests/test_terrain.cpp)
target_link_libraries(test-terrain PRIVATE terrain)
enable_testing()
add_test(NAME terrain-parity COMMAND test-terrain)
add_executable(test-shadow-bounds tests/test_shadow_bounds.cpp)
target_include_directories(test-shadow-bounds PRIVATE src)
target_compile_options(test-shadow-bounds PRIVATE -UNDEBUG -Wall -Wextra)
add_test(NAME shadow-bounds COMMAND test-shadow-bounds ${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/scene_fs.glsl)
add_executable(test-geometry-bounds tests/test_geometry_bounds.cpp)
target_include_directories(test-geometry-bounds PRIVATE src)
target_link_libraries(test-geometry-bounds PRIVATE terrain)
target_compile_options(test-geometry-bounds PRIVATE -UNDEBUG -Wall -Wextra)
add_test(NAME geometry-bounds COMMAND test-geometry-bounds ${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/scene_fs.glsl)
add_library(source-systems src/game.cpp src/world.cpp src/pose.cpp src/particles.cpp src/audio.cpp src/runtime.cpp src/developer_overlay.cpp)
target_include_directories(source-systems PUBLIC src)
target_link_libraries(source-systems PUBLIC terrain PkgConfig::SDL2 Threads::Threads)
target_compile_options(source-systems PRIVATE -Wall -Wextra)
foreach(name gameplay gameplay_world gameplay_pose integration developer_overlay developer_controls)
 add_executable(test-${name} tests/test_${name}.cpp)
 target_link_libraries(test-${name} PRIVATE source-systems)
 add_test(NAME ${name} COMMAND test-${name})
endforeach()
add_executable(test-audio-pcm tests/test_audio_pcm.cpp)
target_include_directories(test-audio-pcm PRIVATE src)
target_link_libraries(test-audio-pcm PRIVATE Threads::Threads PkgConfig::SDL2)
add_test(NAME audio-pcm COMMAND test-audio-pcm)
set_tests_properties(audio-pcm PROPERTIES TIMEOUT 20)
add_executable(test-audio-worker tests/test_audio_worker.cpp)
target_include_directories(test-audio-worker PRIVATE src)
target_link_libraries(test-audio-worker PRIVATE Threads::Threads PkgConfig::SDL2)
add_test(NAME audio-worker COMMAND test-audio-worker)
set_tests_properties(audio-worker PROPERTIES TIMEOUT 20)
add_executable(test-audio tests/test_audio.cpp)
target_link_libraries(test-audio PRIVATE source-systems)
add_test(NAME audio-source COMMAND test-audio)
set_tests_properties(audio-source PROPERTIES TIMEOUT 180)
add_executable(test-audio-diagnostics tests/test_audio_diagnostics.cpp)
target_link_libraries(test-audio-diagnostics PRIVATE source-systems)
add_test(NAME audio-diagnostics COMMAND test-audio-diagnostics)
add_executable(test-audio-lifecycle tests/test_audio_lifecycle.cpp)
target_link_libraries(test-audio-lifecycle PRIVATE source-systems)
add_test(NAME audio-lifecycle COMMAND test-audio-lifecycle)
set_tests_properties(audio-lifecycle PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy" TIMEOUT 30)
add_executable(falllinie-spike src/main.cpp src/renderer.cpp src/game_renderer.cpp src/profiler.cpp src/paired_clocks.cpp)
target_link_libraries(falllinie-spike PRIVATE source-systems)
target_compile_options(falllinie-spike PRIVATE -Wall -Wextra)
add_executable(test-profiler tests/test_profiler.cpp src/profiler.cpp src/paired_clocks.cpp)
target_include_directories(test-profiler PRIVATE src)
target_link_libraries(test-profiler PRIVATE PkgConfig::SDL2)
# Keep lifecycle checks active even in Release builds.
target_compile_options(test-profiler PRIVATE -UNDEBUG -Wall -Wextra)
add_test(NAME profiler COMMAND test-profiler ${CMAKE_CURRENT_BINARY_DIR}/profiler-test)
add_executable(test-paired-clocks tests/test_paired_clocks.cpp src/profiler.cpp src/paired_clocks.cpp)
target_include_directories(test-paired-clocks PRIVATE src)
target_link_libraries(test-paired-clocks PRIVATE PkgConfig::SDL2)
target_compile_options(test-paired-clocks PRIVATE -UNDEBUG -Wall -Wextra)
add_test(NAME paired-clocks COMMAND test-paired-clocks ${CMAKE_CURRENT_BINARY_DIR}/paired-clock-test)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 set(PAIRED_SCRIPTS ${CMAKE_CURRENT_SOURCE_DIR}/ttmp/2026/10/13/FALLLINIE-TRACE-005--cpu-gpu-driver-and-scheduler-attribution-with-perf-ftrace-and-ebpf/scripts/paired-clocks)
 add_test(NAME paired-clock-analyzer COMMAND ${Python3_EXECUTABLE} ${PAIRED_SCRIPTS}/test_analyze.py ${CMAKE_CURRENT_BINARY_DIR}/paired-clock-test)
 set_tests_properties(paired-clocks PROPERTIES FIXTURES_SETUP paired-clock-data)
 set_tests_properties(paired-clock-analyzer PROPERTIES FIXTURES_REQUIRED paired-clock-data)
