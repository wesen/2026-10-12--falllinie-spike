cmake_minimum_required(VERSION 3.16)
project(falllinie_spike LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
find_package(Threads REQUIRED)
add_library(terrain src/terrain.cpp)
target_include_directories(terrain PUBLIC src)
target_compile_options(terrain PRIVATE -Wall -Wextra)
add_executable(test-terrain tests/test_terrain.cpp)
target_link_libraries(test-terrain PRIVATE terrain)
enable_testing()
add_test(NAME terrain-parity COMMAND test-terrain)
add_executable(falllinie-spike src/main.cpp src/renderer.cpp)
target_link_libraries(falllinie-spike PRIVATE terrain)
target_link_libraries(falllinie-spike PRIVATE PkgConfig::SDL2 Threads::Threads)
target_compile_options(falllinie-spike PRIVATE -Wall -Wextra)
